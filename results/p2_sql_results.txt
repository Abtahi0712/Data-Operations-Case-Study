UPDATE p2_encounters p SET reason =
  CASE
    WHEN p.status = 'Imported' THEN 'Imported'
    WHEN p.has_flag THEN 'Non-billable flag code (NORCM/SP/TOS)'
    WHEN EXISTS (SELECT 1 FROM db_only o
                 WHERE o.patient = p.patient AND o.dos = p.dos)
         THEN 'Rendering provider mismatch'
    WHEN EXISTS (SELECT 1 FROM db_only o
                 WHERE o.patient = p.patient AND o.provider = p.provider AND ABS(o.dos - p.dos) <= 7)
         THEN 'Date of service mismatch'
    WHEN p.patient IN (SELECT patient FROM p2_encounters
                       WHERE status = 'Missing' AND NOT has_flag
                       GROUP BY patient HAVING COUNT(*) >= 3)
         THEN 'Patient-level block (run of visits)'
    ELSE 'Isolated gap (no pattern)'
  END;
